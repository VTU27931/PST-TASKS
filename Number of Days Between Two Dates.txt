class Solution {
    public int daysBetweenDates(String a, String b) {
        return Math.abs(days(a) - days(b));
    }

    int days(String s) {
        int y = Integer.parseInt(s.substring(0,4));
        int m = Integer.parseInt(s.substring(5,7));
        int d = Integer.parseInt(s.substring(8));

        int[] md = {31,28,31,30,31,30,31,31,30,31,30,31};
        int res = d;

        for (int i = 0; i < m - 1; i++)
            res += md[i];

        if (m > 2 && (y%400 == 0 || (y%4 == 0 && y%100 != 0)))
            res++;

        res += 365 * (y - 1971) + (y - 1969) / 4 - (y - 1901) / 100 + (y - 1701) / 400;

        return res;
    }
}
