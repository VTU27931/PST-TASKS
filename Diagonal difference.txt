import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[][] a = new int[n][n];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                a[i][j] = sc.nextInt();
            }
        }

        int sum1 = 0, sum2 = 0;

        for (int i = 0; i < n; i++) {
            sum1 += a[i][i];
            sum2 += a[i][n - 1 - i];
        }

        System.out.println(Math.abs(sum1 - sum2));
    }
}
